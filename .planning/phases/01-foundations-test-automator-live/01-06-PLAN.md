**2. `src/app/api/tools/test-automator/route.ts`** — the Node-runtime POST handler.

```typescript
import 'server-only';
import { NextRequest } from 'next/server';
import { anthropic, MODEL } from '@/lib/anthropic';
import { runTool, getClientIp } from '@/lib/run-tool';
import { testAutomatorTool } from '@/lib/tools/test-automator-definition';

// FRAME-09 + RESEARCH Pitfall 7 — Vercel Hobby gives Node 60s; Edge only 25s. Explicit always.
export const runtime = 'nodejs';
export const maxDuration = 60;

export async function POST(req: NextRequest) {
  const ip = getClientIp(req);

  // Plan-05 gate: validation + size cap + rate limit
  const gate = await runTool({ tool: testAutomatorTool, request: req, ip });
  if (!gate.ok) {
    return new Response(JSON.stringify(gate.body), {
      status: gate.status,
      headers: gate.headers ?? { 'Content-Type': 'application/json' },
    });
  }

  // Stream the Anthropic response as SSE
  const encoder = new TextEncoder();
  const stream = new ReadableStream({
    async start(controller) {
      try {
        const anthropicStream = anthropic.messages.stream({
          model: MODEL,
          max_tokens: 4096,
          system: testAutomatorTool.systemPrompt,
          messages: [{ role: 'user', content: testAutomatorTool.buildPrompt(gate.input) }],
        });

        for await (const event of anthropicStream) {
          if (event.type === 'content_block_delta' && event.delta.type === 'text_delta') {
            controller.enqueue(
              encoder.encode(`data: ${JSON.stringify({ text: event.delta.text })}\n\n`)
            );
          }
        }
        controller.enqueue(encoder.encode('data: [DONE]\n\n'));
        controller.close();
      } catch (err) {
        // Map model/network/SDK failures to a single safe message per UI-SPEC error states.
        const errorMsg = 'Claude returned an error. Try again — your input was not charged.';
        controller.enqueue(
          encoder.encode(`data: ${JSON.stringify({ error: errorMsg })}\n\n`)
        );
        controller.close();
      }
    },
  });

  return new Response(stream, {
    status: 200,
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-store',
      'X-Content-Type-Options': 'nosniff',
      'X-RateLimit-Limit': String(gate.rateLimit.limit),
      'X-RateLimit-Remaining': String(gate.rateLimit.remaining),
      'X-RateLimit-Reset': String(gate.rateLimit.reset),
    },
  });
}
```

```typescript
async function generate(currentInput: string, currentMode: 'url' | 'story') {
  setGenerating(true);
  setFirstTokenReceived(false);
  setCode({ playwright: '', pytest: '' });
  setRationale('');
  setError(null);
  setHighlighted(null);

  // Client-side timeout (UI-SPEC: "Request timed out" after 30s with no first token)
  const timeoutId = setTimeout(() => {
    if (!firstTokenReceived) {
      setError('Request timed out. Try a shorter user story or a simpler page URL.');
      setGenerating(false);
    }
  }, 30000);

  let response: Response;
  try {
    response = await fetch('/api/tools/test-automator', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ input: currentInput, mode: currentMode }),
    });
  } catch {
    clearTimeout(timeoutId);
    setError('Claude returned an error. Try again — your input was not charged.');
    setGenerating(false);
    return;
  }

  if (!response.ok) {
    clearTimeout(timeoutId);
    if (response.status === 429) {
      const retryAfter = response.headers.get('Retry-After') ?? '60';
      setError(`Rate limit reached — try again in ${retryAfter}s.`);
    } else if (response.status === 413) {
      const body = await response.json();
      setError(typeof body.error === 'string' ? body.error : `Input exceeds the 4000 character limit. Please shorten your description.`);
    } else if (response.status === 422) {
      setError('Please describe the user story');
    } else {
      setError('Claude returned an error. Try again — your input was not charged.');
    }
    setGenerating(false);
    return;
  }

  // SSE read loop with buffer (Pitfall 4 — handle chunk boundaries mid-message)
  const reader = response.body!.getReader();
  const decoder = new TextDecoder();
  let buffer = '';
  let currentSection: 'playwright' | 'pytest' | 'rationale' = 'playwright';

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    const messages = buffer.split('\n\n');
    buffer = messages.pop() ?? '';
    for (const msg of messages) {
      const line = msg.trim();
      if (!line.startsWith('data: ')) continue;
      const data = line.slice(6);
      if (data === '[DONE]') {
        clearTimeout(timeoutId);
        // After completion: apply Shiki highlighting in background
        try {
          const { getHighlighter } = await import('@/lib/highlighter');
          const h = await getHighlighter();
          setHighlighted({
            playwright: h.codeToHtml(code.playwright, { lang: 'typescript', theme: 'github-dark' }),
            pytest: h.codeToHtml(code.pytest, { lang: 'python', theme: 'github-dark' }),
          });
          // CRITICAL: Even though we have HTML here, we will NOT use dangerouslySetInnerHTML.
          // Instead, store it and use a different render path — see render section.
        } catch {
          // Shiki failure is non-fatal; fall back to plain pre/code
        }
        setGenerating(false);
        return;
      }
      let parsed: { text?: string; error?: string };
      try {
        parsed = JSON.parse(data);
      } catch {
        continue;
      }
      if (parsed.error) {
        clearTimeout(timeoutId);
        setError(parsed.error);
        setGenerating(false);
        return;
      }
      if (parsed.text) {
        if (!firstTokenReceived) {
          setFirstTokenReceived(true);
        }
        let text = parsed.text;
        // Route text by delimiter (delimiters can arrive mid-chunk; check substrings)
        while (text.length > 0) {
          if (currentSection === 'playwright' && text.includes('---PYTEST---')) {
            const [before, ...rest] = text.split('---PYTEST---');
            setCode((prev) => ({ ...prev, playwright: prev.playwright + before }));
            currentSection = 'pytest';
            text = rest.join('---PYTEST---');
            continue;
          }
          if (currentSection === 'pytest' && text.includes('---RATIONALE---')) {
            const [before, ...rest] = text.split('---RATIONALE---');
            setCode((prev) => ({ ...prev, pytest: prev.pytest + before }));
            currentSection = 'rationale';
            text = rest.join('---RATIONALE---');
            continue;
          }
          // No more delimiters — append all remaining text to current section
          if (currentSection === 'playwright') {
            setCode((prev) => ({ ...prev, playwright: prev.playwright + text }));
          } else if (currentSection === 'pytest') {
            setCode((prev) => ({ ...prev, pytest: prev.pytest + text }));
          } else {
            setRationale((prev) => prev + text);
          }
          text = '';
        }
      }
    }
  }
  clearTimeout(timeoutId);
  setGenerating(false);
}
```

**XSS-safe Shiki rendering decision (per T-1-02):**
The Shiki `codeToHtml` call above produces an HTML string. **We will NOT render this via `dangerouslySetInnerHTML`** (CLAUDE.md ban + T-1-02). Two safer paths — pick ONE for Phase 1:

- **Option A (RECOMMENDED for Phase 1 minimal scope):** Skip the `codeToHtml` post-processing entirely. Always render code as plain `<pre><code>{code}</code></pre>`. Apply CSS-only highlighting via the IDE's existing `tk-*` syntax token classes by running the code through `@/lib/syntax-highlighter`'s `tokenize()` after the stream completes (for typescript we'd need to extend the tokenizer or just render mono-color — acceptable given AUTO-05 just requires "syntax-highlighted" which is satisfied by the consistent mono font even if uncolored; though colored is better).
- **Option B (cleaner, slightly more work):** Use Shiki's `codeToTokensBase` (returns a token array) and map each token to a React `<span>` with the appropriate token color from `tokens[i].color`. No HTML strings, no dangerouslySetInnerHTML, full Shiki fidelity.

**Choose Option B.** Implementation:
```typescript
// Helper component — Tokenized code renderer (replaces dangerouslySetInnerHTML pattern)
function ShikiTokens({ code, lang }: { code: string; lang: 'typescript' | 'python' }) {
  const [lines, setLines] = useState<Array<Array<{ content: string; color?: string }>> | null>(null);
  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const { getHighlighter } = await import('@/lib/highlighter');
        const h = await getHighlighter();
        const result = h.codeToTokensBase(code, { lang, theme: 'github-dark' });
        // codeToTokensBase returns ThemedToken[][] (lines of tokens)
        if (!cancelled) setLines(result as any);
      } catch {
        if (!cancelled) setLines(null);
      }
    })();
    return () => { cancelled = true; };
  }, [code, lang]);

  if (!lines) {
    // Plain rendering during/before highlighting
    return <pre className="font-mono text-[13px] leading-[1.65] p-4 overflow-auto"><code>{code}</code></pre>;
  }
  return (
    <pre className="font-mono text-[13px] leading-[1.65] p-4 overflow-auto">
      <code>
        {lines.map((line, i) => (
          <div key={i}>
            {line.map((tok, j) => (
              <span key={j} style={tok.color ? { color: tok.color } : undefined}>{tok.content}</span>
            ))}
            {'\n'}
          </div>
        ))}
      </code>
    </pre>
  );
}
```
