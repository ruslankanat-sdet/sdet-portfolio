Create `src/lib/run-tool.ts`. This is the **shared server-side wrapper** that route handlers compose. The route handler stays thin: it constructs a `tool` object (or imports one), calls `runTool({ tool, request, ip })`, and:
- if `result.ok === true`: opens its SSE stream / runs `messages.stream()` and returns a `Response(stream, { headers })`
- if `result.ok === false`: returns `new Response(JSON.stringify(result.body), { status: result.status, headers: result.headers })`

The wrapper itself does NOT call Anthropic — that's the route handler's job because stream lifecycle is route-specific (SSE message framing, controller close timing, etc.). Keeping `runTool` Anthropic-free also makes Phase 2's structured-output tools reuse it without modification.

**File contents:**

```typescript
import 'server-only';
import type { z } from 'zod';
import type { Tool } from './tool';
import { ratelimitPerTool, ratelimitGlobal } from './ratelimit';

export type RunToolResult<TInput> =
  | {
      ok: true;
      input: TInput;
      identifier: string;
      rateLimit: { limit: number; remaining: number; reset: number };
    }
  | {
      ok: false;
      status: number;
      body: { error: string | Record<string, unknown> };
      headers?: Record<string, string>;
    };

/**
 * Server-side tool execution gate. FRAME-05.
 *
 * Pipeline (each step short-circuits on failure):
 *   1. Parse JSON body                  → 400 on parse error
 *   2. Validate via tool.inputSchema    → 422 on Zod fail
 *   3. Enforce inputCharLimit on the primary text field → 413 on overflow (FRAME-11)
 *   4. Sliding-window rate limit (per-tool 10/60s + global 30/60s) → 429 + Retry-After (FRAME-06, FRAME-07)
 *   5. ok: true — caller proceeds with the LLM call
 *
 * Never throws. All errors are mapped to a typed result.
 */
export async function runTool<TInput>(args: {
  tool: Tool<TInput>;
  request: Request;
  ip: string;
}): Promise<RunToolResult<TInput>> {
  const { tool, request, ip } = args;

  // 1. Parse JSON body
  let body: unknown;
  try {
    body = await request.json();
  } catch {
    return {
      ok: false,
      status: 400,
      body: { error: 'Invalid JSON body' },
    };
  }

  // 2. Zod validation
  const parsed = tool.inputSchema.safeParse(body);
  if (!parsed.success) {
    return {
      ok: false,
      status: 422,
      body: { error: parsed.error.flatten() },
    };
  }

  const input = parsed.data;

  // 3. Input size cap (FRAME-11) — check the `input` field of the validated body if it exists.
  //    All Phase 1+ tools have a string `input` field (Test Automator: input; future tools may differ).
  //    Tools can override by setting inputCharLimit to a smaller value — but we ALWAYS enforce.
  const primaryText = extractPrimaryText(input);
  if (primaryText !== null && primaryText.length > tool.inputCharLimit) {
    return {
      ok: false,
      status: 413,
      body: {
        error: `Input exceeds the ${tool.inputCharLimit} character limit. Please shorten your description.`,
      },
    };
  }

  // 4. Two-tier rate limit
  const identifier = `${tool.id}:${ip}`;
  const [perToolResult, globalResult] = await Promise.all([
    ratelimitPerTool.limit(identifier),
    ratelimitGlobal.limit(ip),
  ]);

  if (!perToolResult.success || !globalResult.success) {
    // Compute the soonest reset between the two limits.
    const reset = Math.min(perToolResult.reset, globalResult.reset);
    const retryAfter = Math.max(1, Math.ceil((reset - Date.now()) / 1000));

    return {
      ok: false,
      status: 429,
      body: {
        error: `Rate limit reached — try again in ${retryAfter}s.`,
      },
      headers: {
        'Retry-After': String(retryAfter),
        'X-RateLimit-Limit': String(perToolResult.limit),
        'X-RateLimit-Remaining': String(perToolResult.remaining),
        'X-RateLimit-Reset': String(reset),
        'Content-Type': 'application/json',
      },
    };
  }

  // 5. Green path
  return {
    ok: true,
    input,
    identifier,
    rateLimit: {
      limit: perToolResult.limit,
      remaining: perToolResult.remaining,
      reset: perToolResult.reset,
    },
  };
}

/**
 * Best-effort extraction of the primary text field for FRAME-11 size cap enforcement.
 * Tools that don't use a flat `{ input: string }` shape should override their schema's `.max()` directly.
 */
function extractPrimaryText(input: unknown): string | null {
  if (input && typeof input === 'object' && 'input' in input) {
    const v = (input as Record<string, unknown>).input;
    if (typeof v === 'string') return v;
  }
  return null;
}

/**
 * Helper to extract a usable IP identifier from the incoming request.
 * Vercel sets x-forwarded-for reliably; x-real-ip is a fallback for some edge configurations.
 * 127.0.0.1 is the local-dev fallback (acceptable per RESEARCH "Don't Hand-Roll" — standard pattern).
 */
export function getClientIp(request: Request): string {
  const xff = request.headers.get('x-forwarded-for');
  if (xff) {
    // x-forwarded-for can be comma-separated (multiple proxies); take the first.
    return xff.split(',')[0].trim();
  }
  const xri = request.headers.get('x-real-ip');
  if (xri) return xri.trim();
  return '127.0.0.1';
}
```

**Notes on the design:**
- The wrapper does NOT swallow Anthropic errors — those happen INSIDE the stream after `runTool` returns ok:true. Plan 06's route handler is responsible for sending an SSE `{error}` event and closing the controller. UI-SPEC error copy `"Claude returned an error. Try again — your input was not charged."` is enforced in the route handler, not here.
